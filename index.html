const SS_ID = "MASUKKAN_ID_SPREADSHEET_ANDA_DI_SINI"; function doGet(e) { try { var ss = SpreadsheetApp.openById(SS_ID); var sheetProduk = ss.getSheetByName("Produk"); var dataProduk = sheetProduk.getDataRange().getValues(); var produk = []; for (var i = 1; i < dataProduk.length; i++) { if (dataProduk[i][0]) { produk.push({ nama: String(dataProduk[i][0]), modalKg: Number(dataProduk[i][1]) || 0, // Harga modal per 1 Kg jualStandar: Number(dataProduk[i][2]) || 0 // Patokan harga jual }); } } return keluarJSON({ status: "success", produk: produk }); } catch(err) { return keluarJSON({ status: "error", message: err.toString() }); } } function doPost(e) { try { var ss = SpreadsheetApp.openById(SS_ID); var sheetTransaksi = ss.getSheetByName("Transaksi"); var data = JSON.parse(e.postData.contents); // Konversi satuan ke Kg untuk hitung modal // Jika satuan 'Ons', maka Qty dibagi 10. Jika 'Kg', Qty tetap. let qtyKg = data.satuan === "Ons" ? data.qty / 10 : data.qty; let totalModal = qtyKg * data.modalKg; let totalOmzet = data.totalOmzet; let totalMargin = totalOmzet - totalModal; // Keuntungan bersih // Simpan ke tab Transaksi sesuai urutan kolom baru sheetTransaksi.appendRow([ new Date(), // Tanggal & Waktu data.namaCust, // Nama Customer data.noWa, // No WA data.namaBarang, // Nama Barang data.qty + " " + data.satuan, // Jumlah Beli (Contoh: 3 Ons) totalOmzet, // Total Omzet totalModal, // Total Modal totalMargin // Margin (Keuntungan) ]); return keluarJSON({ status: "OK", margin: totalMargin }); } catch(err) { return keluarJSON({ status: "ERROR", message: err.toString() }); } } function keluarJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doOptions(e) { return ContentService.createTextOutput("") .setMimeType(ContentService.MimeType.TEXT); }
